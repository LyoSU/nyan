# What the crawler, the daemon and the digest need at runtime. Analytics and
# development tooling live in requirements-analytics.txt / requirements-dev.txt,
# so the production image does not carry them.
#
# Lower bounds are the versions this project is actually verified against.
--extra-index-url https://download.pytorch.org/whl/cpu

# Crawler
# 2.13+ specifically: the spider defines the async start() entry point that
# version introduced. Older Scrapy uses the start_requests() fallback.
scrapy >= 2.13
html2text >= 2024.2.26

# Annotator
fasttext-wheel >= 0.9.2
# The pinned ML stack still resolves numpy 1.x; 2.x is untested here.
numpy >= 1.26, < 2.0
transformers >= 4.40
torch >= 2.2
pyonmttok >= 1.37
natasha >= 1.6
pillow >= 10.4
requests >= 2.32

# Clustering
scipy >= 1.11
scikit-learn >= 1.4
joblib >= 1.3
# fasttext-wheel still imports pkg_resources, removed in setuptools 81.
setuptools < 81

# Client
httpx >= 0.27

# Renderer
jinja2 >= 3.1

# Shared
pymongo >= 4.6
tqdm >= 4.66

# Language models
openai >= 2.0, < 3
